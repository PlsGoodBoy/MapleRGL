if(WIN32)
 enable_language(RC)
 execute_process(COMMAND git rev-parse --short HEAD WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" OUTPUT_VARIABLE MAPLERGL_BUILD_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE)
 execute_process(COMMAND git diff --quiet HEAD WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" RESULT_VARIABLE MAPLERGL_WORKTREE_CHANGED)
 if(MAPLERGL_WORKTREE_CHANGED)
  string(APPEND MAPLERGL_BUILD_COMMIT "-dirty")
 endif()

 string(TIMESTAMP MAPLERGL_BUILD_TIME "%Y-%m-%dT%H:%M:%SZ" UTC)
 configure_file(src/build_info.hpp.in generated/build_info.hpp @ONLY)
 add_executable(MapleRGL WIN32 resources/app.rc src/battle_art_renderer.cpp src/skill_visual_renderer.cpp src/main_win32.cpp src/app.cpp src/app_runtime.cpp src/hud_ui.cpp src/camp_ui.cpp src/collection_ui.cpp src/challenge_ui.cpp third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp third_party/imgui/imgui_widgets.cpp third_party/imgui/imgui_tables.cpp third_party/imgui/backends/imgui_impl_win32.cpp third_party/imgui/backends/imgui_impl_dx11.cpp)
 target_include_directories(MapleRGL PRIVATE "${CMAKE_BINARY_DIR}/generated" resources third_party third_party/imgui third_party/imgui/backends)
 target_compile_definitions(MapleRGL PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
 target_link_libraries(MapleRGL PRIVATE maple_core d3d11 dxgi d3dcompiler dwmapi winmm shell32 psapi)
 if(MSVC)
  target_compile_options(MapleRGL PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/utf-8;/W3>")
 endif()
 add_custom_command(TARGET MapleRGL POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:MapleRGL>/assets")
 add_custom_command(TARGET MapleRGL POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "${MAPLERGL_STAGE_DIR}"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:MapleRGL>" "${MAPLERGL_STAGE_DIR}/MapleRGL.exe"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/assets" "${MAPLERGL_STAGE_DIR}/assets")
 install(TARGETS MapleRGL RUNTIME DESTINATION .)
 install(DIRECTORY assets DESTINATION .)
 if(MSVC)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS_SKIP TRUE)
  include(InstallRequiredSystemLibraries)
  list(GET CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS 0 MAPLERGL_RUNTIME_LIBRARY)
  get_filename_component(MAPLERGL_RUNTIME_DIR "${MAPLERGL_RUNTIME_LIBRARY}" DIRECTORY)
  set(MAPLERGL_RUNTIME_DIR "${MAPLERGL_RUNTIME_DIR}" CACHE INTERNAL "VC++ runtime directory for the release script" FORCE)
  add_custom_target(update_release ALL
   COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_SOURCE_DIR}/tools/update_release.ps1"
    -Executable "$<TARGET_FILE:MapleRGL>" -RuntimeDirectory "${MAPLERGL_RUNTIME_DIR}" -Configuration "$<CONFIG>"
   DEPENDS MapleRGL
   COMMENT "Updating the fixed release folder"
   VERBATIM)
 endif()
endif()
